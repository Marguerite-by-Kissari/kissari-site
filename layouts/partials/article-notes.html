{{- /* Global article numbering across body and separately rendered callout notes. Preserve Goldmark IDs/backlinks; only replace visible numbers and add focus targets. */ -}} {{- $html := . | string -}} {{- $references := findRE `]*>[^<]*` $html -}} {{- $numbers := dict -}} {{- $next := 0 -}} {{- range $references -}} {{- $reference := . -}} {{- $id := replaceRE `^.*href="#fn:([^"]+)".*$` `$1` $reference -}} {{- if not (isset $numbers $id) -}} {{- $next = add $next 1 -}} {{- $numbers = merge $numbers (dict $id $next) -}} {{- $old := printf `
  • ` $id -}} {{- $new := printf `
  • %d` $id $id $next $next -}} {{- $html = replace $html $old $new -}} {{- end -}} {{- $number := index $numbers $id -}} {{- $updated := replaceRE `>[^<]*$` (printf `>%d` $number) $reference -}} {{- $updated = replace $updated `role="doc-noteref"` (printf `role="doc-noteref" aria-label="Lire la note %d"` $number) -}} {{- $html = replace $html $reference $updated -}} {{- end -}} {{- $html = replaceRE `` `` $html -}} {{- return ($html | safeHTML) -}}
